44void TimeManagement::incrementDateTime(DateTime &time,
45 const double dtime) noexcept {
46 int nymd = time.ymd;
47 const double hms = time.hms;
48
49 if (dtime == 0.0) {
50 nymd = incrementDateByDay(nymd, -1);
51 nymd = incrementDateByDay(nymd, 1);
52 }
53
54 const int hours = static_cast<int>(hms) / 10000;
55 const int minutes = (static_cast<int>(hms) % 10000) / 100;
56 const double seconds = hms - hours * 10000 - minutes * 100;
57
58 double total_seconds = hours * 3600.0 + minutes * 60.0 + seconds + dtime;
59
60 while (total_seconds >= 86400.0) {
61 total_seconds -= 86400.0;
62 nymd = incrementDateByDay(nymd, 1);
63 }
64
65 while (total_seconds < 0.0) {
66 total_seconds += 86400.0;
67 nymd = incrementDateByDay(nymd, -1);
68 }
69

71 const int f_minutes = (static_cast<int>(total_seconds) % 3600) / 60;
72 const double f_seconds = std::fmod(total_seconds, 60.0);
73
74 time.ymd = nymd;
75 time.hms = f_hours * 10000 + f_minutes * 100 + f_seconds;
76}

78int TimeManagement::incrementDateByDay(const int ymd,
79 const int adjustment) noexcept {
80 int ny = ymd / 10000;
81 int nm = (ymd % 10000) / 100;
82 nm = std::clamp(nm, 1, 12);
83 int nd = (ymd % 100) + adjustment;
84
85 std::array<int, 12> ndpm;
86 if (m_calendarType == CalendarType::ThreeSixtyDay) {
87 ndpm.fill(30);
88 } else {
89 static constexpr std::array<int, 12> standard_ndpm = {
90 31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31};
91 ndpm = standard_ndpm;
92 }
93
94 const auto is_leap = [](const int year, const CalendarType type) noexcept {
95 if (type != CalendarType::Standard)
96 return false;
97 return (year % 400 == 0) || (year % 4 == 0 && year % 100 != 0);
98 };
99
100 const bool leap = is_leap(ny, m_calendarType);
101
102 if (nd == 0) {
103 nm--;
104 if (nm == 0) {
105 nm = 12;
106 ny--;
107 }
108 nd = ndpm[static_cast<size_t>(nm - 1)];
109 if (nm == 2 && leap)
110 nd = 29;

112

115 nd = 1;
116 nm++;
117 if (nm > 12) {
118 nm = 1;
119 ny++;
120 }
121 }
122 }

127double TimeManagement::differenceInSeconds(const DateTime &time1,
128 const DateTime &time2) noexcept {
129 const int ny1 = time1.ymd / 10000;
130 const int nd1 = getDayOfYear(time1.ymd);
131 const int h1 = static_cast<int>(time1.hms) / 10000;
132 const int m1 = (static_cast<int>(time1.hms) % 10000) / 100;
133 const double s1 = std::fmod(time1.hms, 100.0);
134 const double ns1 = h1 * 3600.0 + m1 * 60.0 + s1;
135
136 const int ny2 = time2.ymd / 10000;
137 const int nd2 = getDayOfYear(time2.ymd);
138 const int h2 = static_cast<int>(time2.hms) / 10000;
139 const int m2 = (static_cast<int>(time2.hms) % 10000) / 100;
140 const double s2 = std::fmod(time2.hms, 100.0);
141 const double ns2 = h2 * 3600.0 + m2 * 60.0 + s2;
142
143 int nd = nd2 - nd1;
144
145 if (ny1 != ny2) {
146 const int nst = (ny2 > ny1) ? 1 : -1;
147 int current_ny = ny1;
148 while (current_ny != ny2) {
149 if (nst > 0) {
150 if (m_calendarType == CalendarType::ThreeSixtyDay) {
151 nd += 360;
152 } else if (m_calendarType == CalendarType::NoLeap) {
153 nd += 365;
154 } else {
155 const bool leap = (current_ny % 400 == 0) ||
156 (current_ny % 4 == 0 && current_ny % 100 != 0);
157 nd += leap ? 366 : 365;
158 }
159 current_ny++;
160 } else {
161 current_ny--;
162 if (m_calendarType == CalendarType::ThreeSixtyDay) {
163 nd -= 360;
164 } else if (m_calendarType == CalendarType::NoLeap) {
165 nd -= 365;
166 } else {
167 const bool leap = (current_ny % 400 == 0) ||
168 (current_ny % 4 == 0 && current_ny % 100 != 0);
169 nd -= leap ? 366 : 365;
170 }
171 }
172 }
173 }
174
175 const double ns = ns2 - ns1;
176 return ns + 86400.0 * static_cast<double>(nd);

179int TimeManagement::getDayOfYear(const int ymd) noexcept {
180 const int ny = ymd / 10000;
181 int nm = (ymd % 10000) / 100;
182 int nd = ymd % 100;
183
184 static constexpr std::array<int, 12> standard_ndpm = {31, 28, 31, 30, 31, 30,
185 31, 31, 30, 31, 30, 31};
186
187 const bool leap = (m_calendarType == CalendarType::Standard)
188 ? ((ny % 400 == 0) || (ny % 4 == 0 && ny % 100 != 0))
189 : false;
190
191 if (nm > 2 && leap)
192 nd++;
193
194 while (nm > 1) {
195 nm--;
196 if (m_calendarType == CalendarType::ThreeSixtyDay) {
197 nd += 30;
198 } else {
199 nd += standard_ndpm[static_cast<size_t>(nm - 1)];
200 }
201 }
202

231 const std::span<const int, 8> dateArray, double &julian,

233 const int year = dateArray[0];
234 const int month = dateArray[1];
235 const int day = dateArray[2];
236 const int hour = dateArray[4];
237 const int minute = dateArray[5];

240

244 errorCode = 0;

263void TimeManagement::julianDayToDateArray(const double julian,
264 const std::span<int, 8> dateArray,
265 int &errorCode) noexcept {
266 if (m_calendarType == CalendarType::Standard && julian < 0.0) {
267 errorCode = 1;
268 return;
269 }
270
271 const int ijul_init = static_cast<int>(std::floor(julian));
272 int ijul = ijul_init;
273 double second = (julian - static_cast<double>(ijul)) * 86400.0;
274 const int tz = 0; // UTC
275
276 if (m_calendarType == CalendarType::Standard) {
277 if (second >= 43200.0) {
278 ijul++;
279 second -= 43200.0;
280 } else {
281 second += 43200.0;
282 }
283 }
284
285 if (second >= 86400.0) {
286 ijul++;
287 second -= 86400.0;
288 }
289
290 const int minute_init = static_cast<int>(second / 60.0);
291 const double sec = std::fmod(second, 60.0);
292 const int hour = minute_init / 60;
293 const int minute = minute_init % 60;
294
295 int year, month, day;
296 if (m_calendarType == CalendarType::ThreeSixtyDay) {
297 year = static_cast<int>(julian / 360.0) + 1800;
298 month = (static_cast<int>(julian / 30.0) % 12) + 1;
299 day = (static_cast<int>(julian) % 30) + 1;
300 } else {
301 const int jalpha = static_cast<int>(
302 (static_cast<double>(ijul - 1867216) - 0.25) / 36524.25);
303 const int ja = ijul + 1 + jalpha - static_cast<int>(0.25 * jalpha);
304 const int jb = ja + 1524;
305 const int jc = static_cast<int>(
306 6680.0 + (static_cast<double>(jb - 2439870) - 122.1) / 365.25);
307 const int jd = 365 * jc + static_cast<int>(0.25 * jc);
308 const int je = static_cast<int>(static_cast<double>(jb - jd) / 30.6001);
309 day = jb - jd - static_cast<int>(30.6001 * je);
310 month = je - 1;
311 if (month > 12)
312 month -= 12;
313 year = jc - 4715;
314 if (month > 2)
315 year--;
316 if (year <= 0)
317 year--;
318 }
319
320 dateArray[0] = year;
321 dateArray[1] = month;
322 dateArray[2] = day;
323 dateArray[3] = tz;
324 dateArray[4] = hour;
325 dateArray[5] = minute;
326 dateArray[6] = static_cast<int>(sec);
327 dateArray[7] = static_cast<int>(std::round(std::fmod(sec, 1.0) * 1000.0));
328 errorCode = 0;

369void TimeManagement::parseUnitsToDateArray(const std::string_view units,
370 const std::span<int, 8> dateArray,
371 int &errorCode) noexcept {
372 std::fill(dateArray.begin(), dateArray.end(), 0);
373 errorCode = 1;
374
375 const size_t since_pos = units.find("since ");
376 if (since_pos == std::string_view::npos)
377 return;
378
379 const std::string_view date_part = units.substr(since_pos + 6);
380
381 const auto parse_int = [](const std::string_view s,
382 int &val) -> std::string_view {
383 if (s.empty())
384 return s;
385 const char *first = s.data();
386 const char *last = s.data() + s.size();
387 auto [ptr, ec] = std::from_chars(first, last, val);
388 if (ec != std::errc())
389 return s;
390 const size_t consumed = static_cast<size_t>(ptr - first);
391 if (consumed < s.size() && (s[consumed] == '-' || s[consumed] == ' ' ||
392 s[consumed] == ':' || s[consumed] == 'T')) {
393 return s.substr(consumed + 1);
394 }
395 return s.substr(consumed);
396 };
397
398 int y = 0, m = 0, d = 0;
399 std::string_view remaining = date_part;
400 remaining = parse_int(remaining, y);
401 remaining = parse_int(remaining, m);
402 remaining = parse_int(remaining, d);
403
404 if (y != 0 && m != 0 && d != 0) {
405 dateArray[0] = y;
406 dateArray[1] = m;
407 dateArray[2] = d;
408 errorCode = 0;
409
410 int h = 0, mi = 0, s = 0;
411 remaining = parse_int(remaining, h);
412 remaining = parse_int(remaining, mi);
413 remaining = parse_int(remaining, s);
414
415 dateArray[4] = h;
416 dateArray[5] = mi;
417 dateArray[6] = s;
418 }

463TimeManagement::differenceInDays(const std::span<const int, 8> t1,
464 const std::span<const int, 8> t2) noexcept {
465 if (m_calendarType == CalendarType::ThreeSixtyDay) {
466 const int ad =
467 (t2[0] - t1[0]) * 360 + (t2[1] - t1[1]) * 30 + (t2[2] - t1[2]);
468 const double e1 =
469 3600.0 * t1[4] + 60.0 * (t1[5] - t1[3]) + t1[6] + t1[7] / 1000.0;
470 const double e2 =
471 3600.0 * t2[4] + 60.0 * (t2[5] - t2[3]) + t2[6] + t2[7] / 1000.0;
472 return static_cast<double>(ad) + (e2 - e1) / 86400.0;
473 } else {
474 const int a1 = (14 - t1[1]) / 12;
475 const int b1 = t1[0] + 4800 - a1;
476 const int c1 = t1[1] + 12 * a1 - 3;
477 const int d1_init = t1[2] + (153 * c1 + 2) / 5 + 365 * b1;
478 const int d1 = (m_calendarType == CalendarType::Standard)
479 ? (d1_init + b1 / 4 - b1 / 100 + b1 / 400)
480 : d1_init;
481 const double e1 =
482 3600.0 * t1[4] + 60.0 * (t1[5] - t1[3]) + t1[6] + t1[7] / 1000.0;
483
484 const int a2 = (14 - t2[1]) / 12;
485 const int b2 = t2[0] + 4800 - a2;
486 const int c2 = t2[1] + 12 * a2 - 3;
487 const int d2_init = t2[2] + (153 * c2 + 2) / 5 + 365 * b2;
488 const int d2 = (m_calendarType == CalendarType::Standard)

490 : d2_init;
491 const double e2 =

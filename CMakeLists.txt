#@brief CMake build configuration for WAVEWATCH IV (WW4) Utilities.
#@details This file defines the project structure, dependencies (GTest),
#and build targets for the WW4_Utils library and its tests.

cmake_minimum_required(VERSION 3.20)

# Optional local configuration file generated by the setup tool.
include(${CMAKE_SOURCE_DIR}/ww4_local_config.cmake OPTIONAL)

project(WW4_Utils VERSION 1.0.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# Set default output directory for executables
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)

# Options for strict warnings and sanitizers
option(WW4_STRICT_WARNINGS "Enable strict compiler warnings and treat as errors" OFF)
option(WW4_USE_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)

# Define common compile options for WW4 targets
set(WW4_COMMON_COMPILE_OPTIONS "")
set(WW4_COMMON_LINK_OPTIONS "")

if(MSVC)
    if(WW4_STRICT_WARNINGS)
        list(APPEND WW4_COMMON_COMPILE_OPTIONS /W4 /WX)
    endif()
else()
    list(APPEND WW4_COMMON_COMPILE_OPTIONS -Wall -Wextra -Wpedantic)
    if(WW4_STRICT_WARNINGS)
        list(APPEND WW4_COMMON_COMPILE_OPTIONS -Werror)
        # Suppress unknown warning option errors for LLVM-based compilers (Clang, IntelLLVM)
        if(CMAKE_CXX_COMPILER_ID MATCHES "Clang" OR CMAKE_CXX_COMPILER_ID STREQUAL "IntelLLVM")
            list(APPEND WW4_COMMON_COMPILE_OPTIONS -Wno-error=unknown-warning-option)
        endif()
    endif()
    if(WW4_USE_SANITIZERS)
        list(APPEND WW4_COMMON_COMPILE_OPTIONS -fsanitize=address,undefined)
        list(APPEND WW4_COMMON_LINK_OPTIONS -fsanitize=address,undefined)
    endif()
endif()

# Include directories
include_directories(include)

# Google Test
include(FetchContent)
FetchContent_Declare(
    googletest
    URL https://github.com/google/googletest/archive/refs/tags/v1.14.0.zip
)
# For Windows: Prevent overriding the parent project's compiler/linker settings
set(gtest_force_shared_crt ON CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(googletest)

# Library
add_library(
    ww4_utils
    src/ww4_utils/time_management.cpp
    src/ww4_utils/memory_utils.cpp
    src/ww4_utils/ww4_std_out.cpp
    src/ww4_utils/ww4_logfile.cpp
    src/ww4_utils/ww4_stand_alone_config.cpp
    src/ww4_utils/ww4_run_config.cpp
)
target_include_directories(ww4_utils PUBLIC include)
target_compile_options(ww4_utils PRIVATE ${WW4_COMMON_COMPILE_OPTIONS})
target_link_options(ww4_utils PRIVATE ${WW4_COMMON_LINK_OPTIONS})

add_library(
    ww4_core
    src/ww4_core/w4core_init.cpp
    src/ww4_core/w4core_hom_input.cpp
    src/ww4_core/w4core_wave.cpp
    src/ww4_core/w4core_finl.cpp
)
target_include_directories(ww4_core PUBLIC include)
target_link_libraries(ww4_core PUBLIC ww4_utils)
target_compile_options(ww4_core PRIVATE ${WW4_COMMON_COMPILE_OPTIONS})
target_link_options(ww4_core PRIVATE ${WW4_COMMON_LINK_OPTIONS})

# Executables
add_executable(ww4_stand_alone src/ww4_progs/ww4_stand_alone.cpp)
target_link_libraries(ww4_stand_alone PRIVATE ww4_core ww4_utils)
target_compile_options(ww4_stand_alone PRIVATE ${WW4_COMMON_COMPILE_OPTIONS})
target_link_options(ww4_stand_alone PRIVATE ${WW4_COMMON_LINK_OPTIONS})

# Place the main program in the exe directory
set_target_properties(
    ww4_stand_alone
    PROPERTIES RUNTIME_OUTPUT_DIRECTORY ${CMAKE_SOURCE_DIR}/exe
)

# Tests
enable_testing()

add_executable(test_time_management tests_L1_L2/ww4_utils/test_time_management.cpp)
target_link_libraries(test_time_management PRIVATE ww4_utils GTest::gtest_main)

add_executable(test_memory_utils tests_L1_L2/ww4_utils/test_memory_utils.cpp)
target_link_libraries(test_memory_utils PRIVATE ww4_utils GTest::gtest_main)

add_executable(test_ww4_std_out tests_L1_L2/ww4_utils/test_ww4_std_out.cpp)
target_link_libraries(test_ww4_std_out PRIVATE ww4_utils GTest::gtest_main)

add_executable(test_ww4_logfile tests_L1_L2/ww4_utils/test_ww4_logfile.cpp)
target_link_libraries(test_ww4_logfile PRIVATE ww4_utils GTest::gtest_main)

add_executable(test_ww4_stand_alone_config tests_L1_L2/ww4_utils/test_ww4_stand_alone_config.cpp)
target_link_libraries(test_ww4_stand_alone_config PRIVATE ww4_utils GTest::gtest_main)

add_executable(test_ww4_run_config tests_L1_L2/ww4_utils/test_ww4_run_config.cpp)
target_link_libraries(test_ww4_run_config PRIVATE ww4_utils GTest::gtest_main)

add_executable(test_w4core_time tests_L1_L2/ww4_core/test_w4core_time.cpp)
target_link_libraries(test_w4core_time PRIVATE ww4_core ww4_utils GTest::gtest_main)

add_executable(test_w4core_wave_output tests_L1_L2/ww4_core/test_w4core_wave_output.cpp)
target_link_libraries(test_w4core_wave_output PRIVATE ww4_core ww4_utils GTest::gtest_main)

add_executable(test_w4core_hom_input tests_L1_L2/ww4_core/test_w4core_hom_input.cpp)
target_link_libraries(test_w4core_hom_input PRIVATE ww4_core ww4_utils GTest::gtest_main)

set(WW4_TEST_TARGETS
    test_time_management
    test_memory_utils
    test_ww4_std_out
    test_ww4_logfile
    test_ww4_stand_alone_config
    test_ww4_run_config
    test_w4core_time
    test_w4core_wave_output
    test_w4core_hom_input
)

foreach(test_target IN LISTS WW4_TEST_TARGETS)
    target_compile_options(${test_target} PRIVATE ${WW4_COMMON_COMPILE_OPTIONS})
    target_link_options(${test_target} PRIVATE ${WW4_COMMON_LINK_OPTIONS})
endforeach()

include(GoogleTest)
gtest_discover_tests(test_time_management)
gtest_discover_tests(test_memory_utils)
gtest_discover_tests(test_ww4_std_out)
gtest_discover_tests(test_ww4_logfile)
gtest_discover_tests(test_ww4_stand_alone_config)
gtest_discover_tests(test_ww4_run_config)
gtest_discover_tests(test_w4core_time)
gtest_discover_tests(test_w4core_wave_output)
gtest_discover_tests(test_w4core_hom_input)
